## Creating certificate

cd ~
touch ~/.rnd
mkdir servercertfiles
cd servercertfiles
mkdir -p {public,private,requests}

openssl genrsa --out caprivkey.key 2048

cat caprivkey.key

openssl req -new -x509 \
-days 365 \
-sha256 \
-key caprivkey.key \
-out cacert.pem \
-subj "/CN=admin Root CA"


Fetch public key info
openssl x509 -in ./cacert.pem -noout -pubkey

Fetch subject and serial info from ca cert
openssl x509 -in ./cacert.pem -noout -subject -serial


view as a text
openssl x509 -in ./cacert.pem -noout -text

## create cert for one of the nodes 
openssl genrsa -out private/couchbase.node1.key 2048

cat private/couchbase.node1.key

## gen a csr 
openssl req -new \
-key private/couchbase.node1.key \
-out requests/couchbase.node1.csr \
-subj "/CN=Couchbase Server 01"

## verify csr

openssl req -in ./requests/couchbase.node1.csr \
-noout \
-text \
-verify

## create cert extensions file (sign this CSR with  Root CA)
cat > server.ext <<EOF
basicConstraints=CA:FALSE
subjectKeyIdentifier = hash
authorityKeyIdentifier = keyid,issuer:always
extendedKeyUsage=serverAuth
keyUsage = digitalSignature,keyEncipherment
EOF

cp /server.ext ./server-01.ext.tmp

echo "subjectAltName = IP:172.17.0.2" >> server-01.ext.tmp


## Gen Cert
openssl x509 -req \
-CA cacert.pem \
-CAkey caprivkey.key \
-CAcreateserial -days 365 \
-in requests/couchbase.node1.csr \
-out public/couchbase.node-01.pem \
-extfile server-01.ext.tmp


## view cert
openssl x509 -in ./public/couchbase.node-01.pem \
-noout \
-text
